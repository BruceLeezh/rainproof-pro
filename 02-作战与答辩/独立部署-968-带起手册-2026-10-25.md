## 第 3 步 · 写网格 + 铸演示资金 + 注资 + 喂真值（deployer 还是 operator 时全做完）

```powershell
node setgrid-v4.js --apply      # 10 格人民币网格（分），干跑先看一遍
# 铸演示资金并注资（RDROP：先给 deployer 铸，再 approve + fundPool）
node -e "const{ethers}=require('./node_modules/ethers');require('dotenv').config();
(async()=>{const p=new ethers.JsonRpcProvider(process.env.SEPOLIA_RPC,undefined,{staticNetwork:true});
const w=new ethers.Wallet(process.env.PRIVATE_KEY,p);
const T=new ethers.Contract(process.env.TOKEN_ADDR,require('../03-合约/RainToken.abi.json'),w);
const V=new ethers.Contract(process.env.CONTRACT_ADDRESS,require('../03-合约/RainDeliveryInsuranceV4.abi.json'),w);
const rider=new ethers.Wallet(process.env.RIDER_KEY,p).address;
for(const [to,amt] of [[w.address,'1000000'],[rider,'10000']]) await (await T.mint(to,amt)).wait();
await (await T.approve(V.target,ethers.parseUnits('500000',0))).wait();
await (await V.fundPool(500000)).wait();
console.log('池子(RDROP分)=',await V.poolBalance(),'· rider RDROP=',await T.balanceOf(rider));})().catch(e=>{console.error(e.shortMessage||e.message);process.exit(1)})"
```

## 第 4 步 · 移交 operator（ irreversible，最后做）

```powershell
node -e "const{ethers}=require('./node_modules/ethers');require('dotenv').config();
(async()=>{const p=new ethers.JsonRpcProvider(process.env.SEPOLIA_RPC,undefined,{staticNetwork:true});
const w=new ethers.Wallet(process.env.PRIVATE_KEY,p);
const V=new ethers.Contract(process.env.CONTRACT_ADDRESS,require('../03-合约/RainDeliveryInsuranceV4.abi.json'),w);
const t=await(await V.transferOperator(process.env.MULTISIG_ADDR)).wait();
console.log('transferOperator tx',t.hash,'· 新 operator',await V.operator());
const T=new ethers.Contract(process.env.TOKEN_ADDR,require('../03-合约/RainToken.abi.json'),w);
const t2=await(await T.transferOwner(process.env.MULTISIG_ADDR)).wait();
console.log('RainToken owner 也移交多签 tx',t2.hash);})().catch(e=>{console.error(e.shortMessage||e.message);process.exit(1)})"
```
